\section{Sub-Topic 1: Connecting Metaverse Ethics to Information Ethics
(\textasciitilde300
Words)}\label{sub-topic-1-connecting-metaverse-ethics-to-information-ethics-300-words}

\subsection{Node 1.1: The Metaverse as an embodiment space: Datafication
vs.~Stewardship}\label{node-1.1-the-metaverse-as-an-embodiment-space-datafication-vs.-stewardship}

\begin{itemize}
\item
  \textbf{Provocative Hook}: Is the Metaverse an expansion of human
  freedom, or a raw data prison?
\item
  \textbf{Core Argument \& IE Connection}: Information Ethics (IE) views
  virtual reality as an extension of Floridi's \emph{infosphere}
  {[}@himma2008{]}. Instead of an amoral playground, users enter as
  \emph{homo poieticus}---creative stewards duty-bound to prevent
  informational \emph{entropy} (the degradation of reality and
  truth)\textsuperscript{{[}1{]}}{[}@himma2008{]}. The core ethical
  dilemma is whether the Metaverse fosters human flourishing or
  commodifies human existence into data streams {[}@capurro2006{]}.
\item
  \textbf{Multi-Media Cue}:

  \begin{itemize}
  \item
    3D spatial node embedding a split-screen video comparing Second Life
    virtual land speculation with physical environmental
    degradation\textsuperscript{{[}10{]}}{[}@himma2008{]}.
  \item
    Embedded video essay comparing \emph{Second Life} land speculation
    and data surveillance with real-world corporate monopolies
    {[}@himma2008{]}.
  \end{itemize}
\end{itemize}

\subsection{Node 1.2: The ``NPC-ification'' of Real Humans (Digital
Metaphysics)}\label{node-1.2-the-npc-ification-of-real-humans-digital-metaphysics}

\begin{itemize}
\item
  \textbf{Provocative Hook}: If you can torture hyper-realistic AI
  avatars in VR with zero consequences, how long until you start viewing
  real humans as disposable NPCs?
\item
  \textbf{Core Argument \& IE Connection}: Discarding cliché debates
  about ``AI rights,'' Capurro critiques ``digital metaphysics''---the
  dangerous tendency to equate human beings with mere data streams
  {[}@capurro2006{]}. AI avatars lack noumenal will or embodied
  consciousness {[}@capurro2006{]}. The true danger isn't hurting an
  AI's feelings; it is the erosion of \emph{parrhesia} (truth-telling)
  and human dignity when real people treat the virtual world as a place
  beyond moral
  accountability\textsuperscript{{[}1{]}}{[}@capurro2006{]}.
\item
  \textbf{Multi-Media Cue}: Interactive split-screen node contrasting VR
  harassment logs with clips from \emph{Westworld} / \emph{Detroit:
  Become Human}.
\end{itemize}

\subsection{Node 1.3: The Social Complacency of Multiplayer Games \&
Under The Banality of Virtual
Evil}\label{node-1.3-the-social-complacency-of-multiplayer-games-under-the-banality-of-virtual-evil}

\begin{itemize}
\item
  \textbf{Provocative Hook}: How do video game UIs turn ordinary
  20-somethings into cold, button-pushing virtual war criminals?
\item
  \textbf{Core Argument \& IE Connection}: In MMOGs and strategy games
  (\emph{Defcon}, \emph{EVE Online}), games reward mechanical efficiency
  (points, numbers going up) while stripping away semantic/emotional
  feedback {[}@sicart2008{]}{[}@sicart2009{]}. Sicart calls this the
  ``banality of simulated evil'': when rule engines abstract human
  suffering into UI metrics, players become unthinking bureaucrats of
  virtual destruction {[}@sicart2009{]}.
\item
  \textbf{Multi-Media Cue}: Audio-visual node playing \emph{Defcon}'s
  nuclear strikes (minimalist radar + muffled crying audio) alongside
  real player voice-chats cheering for maximum kill counts
  {[}@sicart2008{]}.
\end{itemize}

\section{Sub-Topic 2: Does Violence in MMOGs Impact Real Life?
(\textasciitilde300
Words)}\label{sub-topic-2-does-violence-in-mmogs-impact-real-life-300-words}

\subsection{Node 2.1: Does Violence in MMOGs Impact Real Life?
(\textasciitilde300
Words)}\label{node-2.1-does-violence-in-mmogs-impact-real-life-300-words}

\begin{itemize}
\item
  \textbf{Provocative Hook}: Crime rates drop as game sales rise---so
  why are we still obsessed with virtual slaughter?
\item
  The ``contamination thesis'' asserts game violence leaks into
  real-world aggression\textsuperscript{{[}23{]}}{[}@goerger2017{]}.
  However, macro-data disproves the ``contamination thesis'' (games do
  not turn players into real-world mass shooters)
  {[}@goerger2017{]}{[}@schulzke2010{]}. However, focusing only on
  macro-crime ignores the micro-lens: \emph{why} players are drawn to
  virtual violence. Schulzke and Ostritsch argue games operate inside a
  ``magic circle'' where virtual actions lack victims
  {[}@ostritsch2017{]}{[}@schulzke2010{]}, but the moral question shifts
  to whether a game \emph{glorifies} depravity or \emph{forces critical
  reflection} {[}@goerger2017{]}{[}@sicart2009{]}.
\item
  \textbf{Multi-Media Cue}: Interactive slider charting video game sales
  against youth violent crime rates over 30 years (showing inverse
  correlation)\textsuperscript{{[}45{]}}{[}@schulzke2010{]}.
\end{itemize}

\subsection{Node 2.2: Satire vs.~Atrocity Porn (The Endorsement
View)}\label{node-2.2-satire-vs.-atrocity-porn-the-endorsement-view}

\begin{itemize}
\item
  \textbf{Provocative Hook}: Why is torturing in \emph{GTA V}
  satirically defensible, but playing \emph{KZ Manager} morally
  abhorrent\textsuperscript{{[}18{]}{[}46{]}}{[}@ostritsch2017{]}?
\item
  \textbf{Core Argument \& Theoretical Connection}: Goerger and
  Ostritsch argue that graphicness alone does not dictate morality;
  context and value integration
  do\textsuperscript{{[}22{]}{[}24{]}}{[}@goerger2017{]}\textsuperscript{{[}47{]}}{[}@ostritsch2017{]}.
  Under Ostritsch's \emph{Endorsement View}, representing evil
  (\emph{GTA V}) is not immoral if framed satirically, whereas endorsing
  an abhorrent worldview (\emph{KZ Manager}, \emph{Hatred}) violates
  moral obligations\textsuperscript{{[}18{]}}{[}@ostritsch2017{]}.
\item
  \textbf{Multi-Media Cue}: Side-by-side gameplay analysis comparing
  \emph{GTA V}'s satire against \emph{Spec Ops: The Line}'s white
  phosphorus scene and banned titles
  {[}@goerger2017{]}{[}@ostritsch2017{]}.
\end{itemize}

\subsection{Node 2.3: Virtue Scaffolding: Games as Moral
Gymnasiums}\label{node-2.3-virtue-scaffolding-games-as-moral-gymnasiums}

\begin{itemize}
\item
  \textbf{Provocative Hook}: Can playing a violent game make you a
  \emph{more} virtuous
  person\textsuperscript{{[}21{]}}{[}@schulzke2010{]}?
\item
  \textbf{Core Argument \& Theoretical Connection}: Wonderly uses Humean
  sentimentalism to argue ultra-violent gaming erodes
  empathy\textsuperscript{{[}27{]}{[}50{]}}{[}@wonderly2008{]}{[}@schulzke2010{]}.
  Rebutting this, Schulzke demonstrates via Kant that virtual attacks
  carry no intent to harm real
  people\textsuperscript{{[}51{]}}{[}@schulzke2010{]}, and via Aristotle
  that choice-engine games (\emph{Fallout 3}, \emph{The Last of Us})
  serve as moral simulators to practice virtuous
  decision-making\textsuperscript{{[}23{]}}{[}@goerger2017{]}\textsuperscript{{[}21{]}{[}52{]}}{[}@schulzke2010{]}.
\item
  \textbf{Multi-Media Cue}: Interactive decision-tree node simulating a
  moral dilemma from \emph{Fallout 3}, showing how moral engines
  scaffold ethical
  reflection\textsuperscript{{[}52{]}}{[}@schulzke2010{]}.
\end{itemize}
